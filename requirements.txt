# ==============================================================================
# BuildSphere AI - Python Dependencies & Local Setup Guide
# ==============================================================================
# Python Version: Python 3.11+ (Python 3.12 recommended)
#
# QUICKSTART (Local Development):
#   1. Create & activate virtual environment:
#      python3.12 -m venv venv
#      source venv/bin/activate       # On Linux/macOS
#      venv\Scripts\activate          # On Windows
#
#   2. Install dependencies:
#      pip install --upgrade pip
#      pip install -r requirements.txt
#
#   3. Environment setup:
#      cp .env.example .env
#
#   4. Database setup (PostgreSQL 15+):
#      docker compose up -d db        # Or run local PostgreSQL (db: construction_db)
#
#   5. Run migrations:
#      cd backend
#      python manage.py migrate
#
#   6. Create admin user:
#      python manage.py create_admin --email admin@buildsphere.com --password ChangeMe#12345
#
#   7. Run automated test suite:
#      pytest -v
#
#   8. Start backend server (Port 8000):
#      python manage.py runserver 0.0.0.0:8000
#
#   9. Start frontend client (Port 3000):
#      cd ../frontend
#      npm install
#      npm run dev
# ==============================================================================
